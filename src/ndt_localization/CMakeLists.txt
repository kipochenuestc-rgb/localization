cmake_minimum_required(VERSION 3.16)
project(ndt_localization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)
add_executable(ndt_localizer ndt_localizer.cpp)
target_include_directories(ndt_localizer PRIVATE ${PCL_INCLUDE_DIRS})
ament_target_dependencies(ndt_localizer rclcpp sensor_msgs nav_msgs std_msgs pcl_conversions)
target_link_libraries(ndt_localizer ${PCL_LIBRARIES})
install(TARGETS ndt_localizer DESTINATION lib/${PROJECT_NAME})
ament_package()
